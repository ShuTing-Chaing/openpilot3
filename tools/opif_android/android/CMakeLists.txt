cmake_minimum_required(VERSION 3.22)
project(opif_android LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

add_library(opif_protocol STATIC native/opif_protocol.cpp)
set_target_properties(opif_protocol PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(opif_protocol PUBLIC native)
target_compile_options(opif_protocol PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(opif_protocol PUBLIC Threads::Threads)

include(CTest)
if(BUILD_TESTING)
  add_executable(opif_protocol_test tests/protocol_test.cpp)
  target_link_libraries(opif_protocol_test PRIVATE opif_protocol)
  add_test(NAME opif_protocol COMMAND opif_protocol_test)
  add_executable(opif_preview_test tests/preview_test.cpp)
  target_compile_options(opif_preview_test PRIVATE -UNDEBUG)
  add_test(NAME opif_preview COMMAND opif_preview_test)
  add_executable(opif_context_cache_test tests/context_cache_test.cpp)
  target_compile_options(opif_context_cache_test PRIVATE -UNDEBUG)
  add_test(NAME opif_context_cache COMMAND opif_context_cache_test)
endif()

option(OPIF_BUILD_SERVER "Build real QNN server (requires Android and ORT)" ${ANDROID})
option(OPIF_BUILD_JNI "Build the Android app JNI library (requires Android and ORT)" OFF)
if(OPIF_BUILD_SERVER OR OPIF_BUILD_JNI)
  if(NOT ANDROID OR NOT CMAKE_ANDROID_ARCH_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Production server requires Android arm64-v8a; host builds are protocol tests only")
  endif()
  set(ORT_ROOT "" CACHE PATH "ORT headers in include/ and Android arm64 libraries in lib/")
  if(NOT EXISTS "${ORT_ROOT}/include/onnxruntime_cxx_api.h" OR
     NOT EXISTS "${ORT_ROOT}/lib/libonnxruntime.so")
    message(FATAL_ERROR "Run scripts/fetch_runtime.py, then set ORT_ROOT")
  endif()
  add_library(onnxruntime SHARED IMPORTED)
  set_target_properties(onnxruntime PROPERTIES
    IMPORTED_LOCATION "${ORT_ROOT}/lib/libonnxruntime.so"
    INTERFACE_INCLUDE_DIRECTORIES "${ORT_ROOT}/include")
  if(OPIF_BUILD_SERVER)
    add_executable(opif-server native/main.cpp native/qnn_runtime.cpp)
    target_compile_options(opif-server PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(opif-server PRIVATE opif_protocol onnxruntime log dl)
    target_link_options(opif-server PRIVATE "-Wl,-z,max-page-size=16384")
  endif()
  if(OPIF_BUILD_JNI)
    add_library(opif_jni SHARED native/jni_bridge.cpp native/qnn_runtime.cpp)
    target_compile_options(opif_jni PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(opif_jni PRIVATE opif_protocol onnxruntime log dl)
    target_link_options(opif_jni PRIVATE "-Wl,-z,max-page-size=16384")
  endif()
endif()
